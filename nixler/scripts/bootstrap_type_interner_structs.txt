
struct TypeNode {
    kind: int64,
    a: int64,
    b: int64,
    c: int64,
    buf_size_is_name: boolean
}

struct ListTypeNode {
    is_nil: boolean,
    head: TypeNode,
    tail: ListTypeNode
}

struct KeyEntry {
    kind: int64,
    a: int64,
    b: int64,
    c: int64,
    buf_size_is_name: boolean,
    id: int64
}

struct ListKeyEntry {
    is_nil: boolean,
    head: KeyEntry,
    tail: ListKeyEntry
}

struct NameEntry {
    lexeme: string,
    id: int64
}

struct ListNameEntry {
    is_nil: boolean,
    head: NameEntry,
    tail: ListNameEntry
}

struct ListInt64Entry {
    is_nil: boolean,
    head: int64,
    tail: ListInt64Entry
}

struct ListStringEntry {
    is_nil: boolean,
    head: string,
    tail: ListStringEntry
}

struct FieldEntry {
    name: string,
    ty: int64
}

struct ListFieldEntry {
    is_nil: boolean,
    head: FieldEntry,
    tail: ListFieldEntry
}

struct TypeInterner {
    nodes: ListTypeNode,
    by_key: ListKeyEntry,
    names: ListNameEntry,
    name_lexemes: ListStringEntry,
    next_id: int64,
    seq_nil_id: int64
}

struct InternResult {
    interner: TypeInterner,
    id: int64
}

struct ParseResult {
    interner: TypeInterner,
    id: int64,
    ok: boolean
}

struct KeyLookupResult {
    by_key: ListKeyEntry,
    id: int64,
    found: boolean
}

struct NameLookupResult {
    names: ListNameEntry,
    id: int64,
    found: boolean
}

struct IdListResult {
    interner: TypeInterner,
    ids: ListInt64Entry,
    ok: boolean
}

struct BracketArgsResult {
    interner: TypeInterner,
    args: ListInt64Entry,
    ok: boolean
}
