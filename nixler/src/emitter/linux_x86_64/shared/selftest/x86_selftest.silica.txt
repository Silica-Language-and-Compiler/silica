// Shared-layer self-test for the linux_x86_64 emitter: prints one assembler unit that uses every
// helper of x86_vr / x86_isa / x86_mem / x86_float at least once, with register-homed and slot-homed
// operands, immediates and the W / S / H views. run_selftest.sh compiles this with the published
// (macOS) compiler, runs it, and assembles the text with clang (--target=x86_64-linux-gnu) and with
// GNU as on nix. The text is not meant to run: it only has to be accepted by both assemblers.

use x86_vr;
use x86_isa;
use x86_mem;
use x86_float;

fn sets_x(n: int64) -> string {
    case n > 28 of {
        true -> "";
        false -> concat(concat(".set ", concat(x86_vr@vr_x_home_name(n), concat(", ", concat(x86_vr@x86_itos(8 * (n - 10)), "\n")))), sets_x(n + 1))
    }
}

fn sets_d(n: int64) -> string {
    case n > 31 of {
        true -> "";
        false -> concat(concat(".set ", concat(x86_vr@vr_d_home_name(n), concat(", ", concat(x86_vr@x86_itos(160 + 8 * (n - 8)), "\n")))), sets_d(n + 1))
    }
}

fn header() -> string {
    h0: string <- "    .intel_syntax noprefix\n    .text\n.set SVR_AREA, 400\n";
    cs: string <- ".set SVR_CS0, 360\n.set SVR_CS1, 368\n.set SVR_CS2, 376\n.set SVR_CS3, 384\n.set SVR_CS4, 392\n";
    concat(concat(h0, concat(sets_x(11), sets_d(8))), concat(cs, "    .globl selftest_fn\nselftest_fn:\n    push rbp\n    mov rbp, rsp\n    sub rsp, SVR_AREA\n"))
}
